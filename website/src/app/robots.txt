import type { MetadataRoute } from 'next'
 
export default function robots(): MetadataRoute.Robots {
  return {
    rules: {
      userAgent: '*',
      allow: '/',
      disallow: '/api/', // Stops bots from wasting resources crawling backend functions
    },
    sitemap: 'https://quickbg.dev/sitemap.xml',
  }
}
